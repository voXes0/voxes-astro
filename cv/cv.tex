%  Design cues taken from the website's About page: warm brown ink, sage-green

%  rounded cards and chips.
%
%  Compile with XeLaTeX or LuaLaTeX (NOT pdfLaTeX), twice (card heights).
%    Overleaf: Menu → Compiler → XeLaTeX
%    Local:    sh cv/build.sh      (builds light + dark into public/)
%  Single build by hand, from inside cv/ (dark = 1, light = 0):
%    xelatex "\def\cvdark{1}\input{cv.tex}"
%
%  Layout is one column, text is real selectable text, and decorations
%  (dots, lines, chips) are drawn graphics, so applicant-tracking software
%  reads it in the correct order.
%  The Persian version is cv-fa.html (see the note at its top).
% ─────────────────────────────────────────────────────────────────────────────

% Small rounded label, e.g. the "Curriculum Vitae" pill above the name.
\newcommand{\cvpill}[1]{%
  \tikz[baseline=(p.base)]\node[fill=pillfill,rounded corners=5pt,
    inner xsep=7pt,inner ysep=3pt,text=accenttext,font=\fontsize{7}{8}\selectfont\bfseries] (p)
    {\addfontfeatures{LetterSpace=8}\MakeUppercase{#1}};}

% Tag chip (technologies, skills).
\newcommand{\cvchip}[1]{%
  \tikz[baseline=(c.base)]\node[fill=chipfill,rounded corners=2.5pt,
    inner xsep=4.2pt,inner ysep=2.2pt,text=ink,font=\fontsize{7.4}{8.5}\selectfont] (c){#1};\hspace{1.4mm}}

% Section heading: tracked caps in accent colour + hairline running to margin.
\newcommand{\cvsection}[1]{%
  \par\vspace{4mm}\noindent
  {\color{accenttext}\fontsize{7.6}{9}\selectfont\bfseries
   \addfontfeatures{LetterSpace=10}\MakeUppercase{#1}}%
  \hspace{3mm}{\color{hair}\leaders\hrule height 3pt depth -2.5pt\hfill}%
  \par\vspace{2.2mm}\nobreak}

% Proficiency dots, 1–5.
\newcommand{\cvdots}[1]{%
  \tikz[baseline=-0.6ex]{%
    \foreach \i in {1,...,5}{%
      \ifnum\i>#1 \fill[hair] (\i*2.7mm,0) circle (0.78mm);%
      \else \fill[accent] (\i*2.7mm,0) circle (0.78mm);\fi}}}

% Bullet list used inside job entries.
\newlist{cvbullets}{itemize}{1}
\setlist[cvbullets]{
  label={\tikz[baseline=-0.34ex]\fill[accent] (0.7mm,0) circle (1.05pt);},
  leftmargin=4.2mm, labelsep=0mm, itemsep=1.2pt, topsep=1.6pt, parsep=0pt, partopsep=0pt,
  font=\small}

% Timeline container: a vertical line down the left edge.
\newtcolorbox{cvtimeline}{coltext=ink,
  enhanced, breakable, frame hidden, interior hidden, boxrule=0pt,
  sharp corners, boxsep=0pt, top=0pt, bottom=0pt, right=0pt, left=5.4mm,
  borderline west={0.7pt}{0pt}{hair}}

% One freelance/job entry.
%   #1 title  #2 client/company · place  #3 period
\newenvironment{cvjob}[3]{%
  \par\vspace{0.6mm}\noindent
  \tikz[overlay,baseline]\fill[accent] (-5.4mm,0.85ex) circle (1.6mm);%
  \tikz[overlay,baseline]\fill[pagebg] (-5.4mm,0.85ex) circle (0.6mm);%
  {\bfseries\fontsize{10.6}{12}\selectfont #1}\hfill
  {\footnotesize\color{muted}#3}\par
  {\small\color{accenttext}\bfseries #2}\par\vspace{0.4mm}%
}{\par\vspace{2.4mm}}

% Project card.
\tcbset{cvcardstyle/.style={coltext=ink, enhanced, colback=cardbg, colframe=hair,
  boxrule=0.7pt, arc=3mm, left=3.4mm, right=3.4mm, top=2.6mm, bottom=2.6mm,
  equal height group=projects}}
\newtcolorbox{cvcard}[1]{cvcardstyle,
  before upper={{\color{muted}\fontsize{6.8}{8}\selectfont\bfseries
    \addfontfeatures{LetterSpace=8}\MakeUppercase{#1}}\par\vspace{1mm}}}
